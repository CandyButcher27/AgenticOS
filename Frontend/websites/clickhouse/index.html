query.sql 
SELECT
    toStartOfHour(event_time) AS hour,
    count() AS requests,
    quantile(0.99)(latency_ms) AS p99
FROM analytics.events
WHERE event_time >= now() - INTERVAL 1 DAY
GROUP BY hour
ORDER BY hour DESC
LIMIT 24;
